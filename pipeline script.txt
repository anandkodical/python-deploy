pipeline {
    agent any

    environment {
        IMAGE_NAME = 'python-flask-app'
        DOCKER_USERNAME = 'anandandy411'
    }

    stages {
        stage('Clone Repository') {
            steps {
                git branch: 'main', url: 'https://github.com/anandkodical/python-deploy.git'
            }
        }

        stage('Build Docker Image') {
            steps {
                script {
                    sh 'docker build -t $IMAGE_NAME .'
                }
            }
        }

        stage('Run Docker Container') {
            steps {
                script {
                    // Stop and remove existing container if running
                    sh '''
                    docker stop $IMAGE_NAME || true
                    docker rm $IMAGE_NAME || true
                    docker run -d -p 5000:5000 --name $IMAGE_NAME $IMAGE_NAME
                    '''
                }
            }
        }
        stage('Cleanup') {
            steps {
                script {
                    sh 'docker image prune -f'
                }
            }
        }
        stage('push to dockerhub'){
            steps{
                script{
                    withCredentials([gitUsernamePassword(credentialsId: 'dockerhub-credentials', gitToolName: 'Default')]) {
                        sh 'docker tag $IMAGE_NAME $DOCKER_USERNAME/$IMAGE_NAME:latest'
                        sh 'docker push $DOCKER_USERNAME/$IMAGE_NAME:latest'
                    }
                    
                }
            }
        }
    }

    post {
        success {
            echo 'App deployed successfully!'
        }
        failure {
            echo 'Deployment failed.'
        }
    }
}
